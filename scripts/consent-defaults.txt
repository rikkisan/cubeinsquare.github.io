    <script>
        // Consent Mode defaults, set before any Google tag can read them (ANALYTICS.md, "Consent").
        // EEA, UK and Switzerland: denied until Google's consent message updates them. Elsewhere: the
        // answer given in the site's own banner, and "declined" also asks AdSense for non-personalized ads.
        (function () {
            const eea = ['AT', 'BE', 'BG', 'CY', 'CZ', 'DE', 'DK', 'EE', 'ES', 'FI', 'FR', 'GR', 'HR', 'HU',
                'IE', 'IT', 'LT', 'LU', 'LV', 'MT', 'NL', 'PL', 'PT', 'RO', 'SE', 'SI', 'SK', 'IS', 'LI', 'NO', 'GB', 'CH'];
            let choice = '';
            try {
                choice = window.localStorage.getItem('cube_cookie_consent') || '';
            } catch (error) {
                choice = '';
            }
            const state = (value) => ({
                ad_storage: value,
                ad_user_data: value,
                ad_personalization: value,
                analytics_storage: value,
                personalization_storage: value,
                functionality_storage: 'granted',
                security_storage: 'granted',
                wait_for_update: 500
            });
            window.dataLayer = window.dataLayer || [];
            window.gtag = window.gtag || function gtag() { window.dataLayer.push(arguments); };
            window.gtag('consent', 'default', { ...state('denied'), region: eea });
            window.gtag('consent', 'default', state(choice === 'accepted' ? 'granted' : 'denied'));
            if (choice === 'declined') (window.adsbygoogle = window.adsbygoogle || []).requestNonPersonalizedAds = 1;
            window.cubeConsentDefaults = true;
        })();
    </script>
